% Artificial Intelligence (AIJ, Elsevier) submission build.
%
%   tectonic -X compile main_aij.tex
%
% This is a wrapper, not a fork: every section, table, figure and generated
% number is the same file the other builds use. Only the page furniture and the
% Elsevier-mandated declarations differ.
%
% Why these class options, against the journal's own instructions:
%
%   elsarticle  AIJ's guide for authors: "You are recommended to use the
%               Elsevier article class elsarticle.cls ... and BibTeX".
%   preprint    AIJ's formatting requirements ask for a single-column layout
%               and warn against "small-font, double-column conference
%               proceeding formats". `preprint` is elsarticle's single-column
%               submission layout with wide margins for annotation.
%   11pt        AIJ sets a minimum 11pt body font.
%   (numbers)   elsarticle defaults to numeric citations; AIJ lists references
%               "in alphabetical order under the first author's name and
%               numbered consecutively by arabic numbers", which is what
%               elsarticle-num produces. Do NOT add the `authoryear` option.
%
% Review is single-anonymized at AIJ -- "Reviewers know the identity of the
% authors" -- so unlike the TMLR and NeurIPS builds this one is named, and the
% artefact link stays in.
\documentclass[preprint,11pt]{elsarticle}

\usepackage[T1]{fontenc}

% natbib is loaded by elsarticle itself; loading it again clashes.
\usepackage[hidelinks]{hyperref}

% Line numbers for reviewers. AIJ asks for a layout that "facilitates margin
% annotations"; this is the other half of that. Delete for the accepted version.
\usepackage{lineno}
\linenumbers

% elsarticle's stock `preprint' block is 5.3in wide, and the generated tables --
% which are built once and shared by all three manuscript builds -- do not fit
% in it at any legible size: the same sources produce 3 overfull boxes at 6.5in
% and 35 at 5.3in. One-inch margins on Letter restore the 6.5in measure, keep
% the single-column 11pt layout AIJ asks for, and still leave a full inch for
% the margin annotations the journal wants room for.
\usepackage[letterpaper,margin=1in]{geometry}

\captionsetup{font=small,labelfont=bf,skip=4pt}
\setlength{\abovecaptionskip}{4pt}
\setlength{\belowcaptionskip}{2pt}

% Compact confidence-interval notation used by every generated table, and by
% the prose, through the facts macros. The other builds set it in \scriptsize,
% which on an 11pt base is 8pt: larger than the 6.5pt the widest appendix
% tables are set in, so the intervals were pushing those tables off the page
% while shrinking the table font did nothing. \smaller is relative, so one
% definition serves an 11pt paragraph and a 6.5pt table cell.
\usepackage{relsize}
\newcommand{\ci}[2]{{\smaller[1]\,[#1,\,#2]}}
% Same 6.5in measure as the TMLR build, but an 11pt rather than a 10pt base, so
% the named sizes all land one notch too large: \scriptsize is 8pt here against
% 7pt there, and the widest tables run 50pt past the margin. Ask for the
% absolute size instead. The journal's 11pt minimum is a body-text rule; tables
% and the verbatim transcripts are set smaller in every Elsevier AI paper.
\newcommand{\tblfont}{\fontsize{7}{8.4}\selectfont}
\newlength{\tblsep}\setlength{\tblsep}{1.5pt}
\newcommand{\tblfontsmall}{\fontsize{6.5}{7.8}\selectfont}
\newcommand{\verbfont}{\fontsize{7}{8.4}\selectfont}
\newlength{\tblsepsmall}\setlength{\tblsepsmall}{2pt}

% Let TeX stretch interword space rather than overhang the margin. The appendix
% is full of unbreakable code identifiers and at 11pt several lines have no
% legal break point at all.
\setlength{\emergencystretch}{3em}

\newcommand{\afail}{a^{\times}}

\newcommand{\copyterm}{\Delta_{\mathrm{copy}}}

\newcommand{\ctxpre}{C}
\newcommand{\todo}[1]{\textcolor{red}{\textbf{[#1]}}}
